def calculate_friedman_h_statistic(model, X, feature_names, n_samples=1000):
    from itertools import combinations
    
    n_features = X.shape[1]
    feature_pairs = list(combinations(range(n_features), 2))
    H_statistics = {}
    feature_medians = np.median(X, axis=0)
    
    for i, j in feature_pairs:
        X_mc = np.zeros((n_samples, n_features))
        
        for col in range(n_features):
            X_mc[:, col] = feature_medians[col]
        
        X_mc[:, i] = np.random.uniform(low=np.min(X[:, i]), high=np.max(X[:, i]), size=n_samples)
        X_mc[:, j] = np.random.uniform(low=np.min(X[:, j]), high=np.max(X[:, j]), size=n_samples)
        
        F_ij = model.predict(X_mc)
        
        X_i = X_mc.copy()
        X_i[:, j] = feature_medians[j]
        F_i = model.predict(X_i)
        
        X_j = X_mc.copy()
        X_j[:, i] = feature_medians[i]
        F_j = model.predict(X_j)
        
        X_0 = X_mc.copy()
        X_0[:, i] = feature_medians[i]
        X_0[:, j] = feature_medians[j]
        F_0 = model.predict(X_0)
        
        interaction_term = F_ij - F_i - F_j + F_0
        
        numerator = np.sum(interaction_term ** 2)
        total_effect = F_ij - F_0
        denominator = np.sum(total_effect ** 2)
        
        if denominator <= 1e-10 or np.isnan(numerator) or np.isnan(denominator):
            H = 0
        else:
            H = numerator / denominator
            H = min(1.0, max(0.0, H))
        
        H_statistics[(i, j)] = H
    
    H_stats_list = []
    for (i, j), value in H_statistics.items():
        H_stats_list.append({
            'Feature1': feature_names[i],
            'Feature2': feature_names[j],
            'H_statistic': value
        })
    
    H_stats_df = pd.DataFrame(H_stats_list)
    H_stats_df = H_stats_df.sort_values(by='H_statistic', ascending=False)
    
    return H_stats_df

def calculate_shap_values(models, X_train, X_test, feature_names):

        shap_values_dict[name] = {
            'values': shap_values,
            'expected_value': explainer.expected_value,
            'explainer': explainer
        }
        
        # Get global feature importance from SHAP
        feature_importance = np.abs(shap_values).mean(0)
        shap_importance = pd.DataFrame({
            'Feature': feature_names,
            'Importance': feature_importance
        }).sort_values('Importance', ascending=False)
        
        shap_values_dict[name]['feature_importance'] = shap_importance
        
        # Calculate SHAP interaction values
        if name == 'XGBoost':  # Only do this for one model due to computational cost
            try:
                # This is computationally intensive - use a smaller sample if needed
                smaller_sample = X_sample[:min(200, len(X_sample))]
                shap_interaction_values = explainer.shap_interaction_values(smaller_sample)
                
                # Store interaction values
                shap_values_dict[name]['interaction_values'] = shap_interaction_values
                
                # Process interaction values
                feature_interactions = []
                for i in range(len(feature_names)):
                    for j in range(i+1, len(feature_names)):
                        # Get mean absolute interaction value
                        interaction_strength = np.abs(shap_interaction_values[:, i, j]).mean()
                        feature_interactions.append({
                            'Feature1': feature_names[i],
                            'Feature2': feature_names[j],
                            'Interaction_Strength': interaction_strength
                        })
                
                # Create interaction DataFrame
                interaction_df = pd.DataFrame(feature_interactions)
                interaction_df = interaction_df.sort_values('Interaction_Strength', ascending=False)
                shap_values_dict[name]['feature_interactions'] = interaction_df
            except Exception as e:
                # If SHAP interaction values fail (they can be memory intensive)
                shap_values_dict[name]['interaction_values'] = None
                shap_values_dict[name]['feature_interactions'] = None
    
    return shap_values_dict

3. `.unique()` —— 把重复的编号去重，只保留不重复的编号列表
4. `np.random.choice(..., size=3)` —— 从这些不重复的编号里，**随机抽 3 个出来**

所以 `random_numbers` 是一个列表，里面装着3个编号，比如 `[7, 42, 15]`。

---

## 然后它被用来干什么？

```python
validation_set = df[df["Number"].isin(random_numbers)]  # 这3个编号对应的行 → 验证集
drop_df = df[~df["Number"].isin(random_numbers)]        # 剩下的所有行 → 训练/测试用
```

就是按编号把数据分成了两拨：

- **`validation_set`**：包含那3个编号的所有数据行，这些数据**完全不会进入模型训练**
- **`drop_df`**：其余编号的所有数据行，后续的 X/y、训练集测试集都是从这堆里来的

---

## 为什么要按编号分，而不是随机抽几行？

**这是这段代码最聪明的地方。**

如果按“行”随机抽样，同一个编号下的数据可能会一部分跑到训练集，一部分跑到测试集——这样模型其实已经“见过”这个编号的数据了，那测试结果就会虚高，这叫**数据泄漏**。

而按“编号”抽，等于保证：**这3组实验的数据，模型从头到尾一次都没见过**。用它来验证模型的泛化能力，结果才真正可信。

## 为什么叫“独热”？先理解名字

“独热”的意思是：**一整排0里，只有一个位置是1（热的）**。就像一排灯，只有“当前这个类别”那一盏亮着，其他全灭。

比如 `Pollutant` 有三种污染物：

| 原始值 | 甲基橙列 | 亚甲基蓝列 | 罗丹明B列 |
|--------|----------|------------|-----------|
| 甲基橙 | **1**    | 0          | 0         |
| 亚甲基蓝 | 0      | **1**      | 0         |
| 罗丹明B | 0       | 0          | **1**     |

每行只有一个“1”是热的，其余都是“0”冷的——这就是“独热”的含义。

---

## 它到底解决了什么问题？

机器学习模型只认识数字，不认识“甲基橙”“亚甲基蓝”这种文字。独热编码的作用就是：**把这个文字列，拆成好几列0/1的开关，让模型能“看懂”类别信息。**

如果没有这一步，模型根本没法处理文字，也没法利用“污染物类型”这个非常重要的特征。

---

## 为什么不用简单的数字来编码？

你可能会想：直接给污染物编号不行吗？比如：

- 甲基橙 = 1
- 亚甲基蓝 = 2
- 罗丹明B = 3

这样用数字表示，**你感觉行，但模型会误解**——它会以为“3比1大”，还会以为“1、2、3之间有顺序关系”，甚至算出“亚甲基蓝+罗丹明B=某种颜色”这种无厘头的操作。这就是**标签编码的隐藏陷阱**。

而独热编码避免了这个问题：每个类别都是独立的维度，互不影响，模型无法在类别之间强行建立大小顺序。它只会在数据中出现某类时，把那条对应的开关打开。

## 一、独热编码的算法到底是什么？

一句话原理：**把“一个类别列”拆成“一堆0/1开关列”，每行只点亮一个开关。**

但计算机具体是怎么做的呢？分四步：

### 第1步：收集所有不重复的类别

```python
unique_values = df["Pollutant"].unique()
```

比如得到：`array(['甲基橙', '亚甲基蓝', '罗丹明B'], dtype=object)`

这就是“候选灯位”——有多少个类别，后面就建多少列。

### 第2步：为每个类别创建一列0/1

核心操作其实是一个**布尔比较**：

```python
for category in unique_values:
    new_col = (df["Pollutant"] == category).astype(int)
```

`df["Pollutant"] == category` 会对每一行做判断：是→`True`，不是→`False`；然后 `.astype(int)` 把布尔值变成 1 和 0。

这一下就把“甲基橙”变成了一列：是甲基橙的就是1，不是的就是0。

### 第3步：把这些新列拼回原表

```python
encoded_df = pd.concat([df, 所有新列], axis=1)
```

一个新列对应一个类别，最后拼成一整张大表。

### 第4步：删掉原来的文字列

因为文字本身已经没用了，留着反而干扰。

## 二、用一个“手算版”来理解

假设原始数据是：

```
Pollutant
甲基橙
亚甲基蓝
甲基橙
```

**第1步**：找到类别 → `[甲基橙, 亚甲基蓝]`

**第2步**：逐类别比较

```
Pollutant == 甲基橙   →  [True, False, True]   →  [1, 0, 1]
Pollutant == 亚甲基蓝 →  [False, True, False]  →  [0, 1, 0]
```

**第3步**：拼起来

```

1       0
0       1
1       0
```

就成了一段“每行只有一个1、其余都是0”的经典独热码。

## 三、`isin` 的算法又是什么？

你之前问过 `isin`，它的底层逻辑也是类似思想，但方向相反——**查成员**。

```python
df["Number"].isin(random_numbers)
```

它的内部算法大致是：

1. 把 `random_numbers` 转成一个**哈希集合（hash set）**——就是对每个编号做一次哈希映射，存进一张“快速查找表”
2. 然后遍历 `df["Number"]` 里的每一个值
3. 每个值去哈希集合里查一下：在 → `True`；不在 → `False`

为什么用哈希集合？因为**查找速度是O(1)**——不管集合里有多少个元素，一次定位就能找到。如果不用哈希，用普通列表一个个比，那就是O(n)甚至O(n²)，数据一多就慢了。

所以 `isin` 真正的算法本质是：**哈希集合 + 逐行查表**。

| 操作 | 底层算法核心 |
|------|-------------|
| `dropna()` | 遍历检查 + 过滤 |
| `unique()` | 哈希去重 |
| `isin()` | 哈希查表 |
| `get_dummies()` | 逐类别布尔比较 + 拼接 |
| `drop(columns=)` | 按列名索引删除 |
| `KNNImputer` | 距离计算 + K近邻投票 |
| `train_test_split` | 随机采样 + 索引切分 |
